# The interpreter's smoke script: hydra_uitest --script tests/ui/scripts/smoke.txt
# It runs most verbs once on the testdata/input corpus and passes when every
# line does. Verb table: docs/agents/ui-testing.md.
timeout 30
click Scan library
wait-idle
click //Scanning charts/Continue
wait-idle
wait 0.1
state
expect-text Burnout
dump Hydra
dump Hydra | Scan
text
text | Burnout
rect Scan library
key Escape
key Ctrl+F
expect-text Burnout
# A bare bar at the end of a line is a bar with nothing after it: type clears
# the field, and text and dump filter on nothing, which keeps every line.
type **/##search | zzqx
expect-text No charts match your search.
type **/##search |
expect-not-text No charts match your search.
expect-text Burnout
text |
dump Hydra |
reset
state
expect-not-text Burnout
quit
# quit ends the script with a pass, so this line never runs. If it did, the
# timeout verb would refuse it.
timeout 90s
