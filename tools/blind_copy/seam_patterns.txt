# Seam forms grade_seams.py searches a slice for. A seam is a place where a
# test swaps a production function, or a production object's field, for its
# own. Only forms known today are here; a test that fakes its input another
# way is not found by this file.
#
# One form per line: label | regex | example file | example line.
# The example line is a whole line, stripped, of that file in the main
# checkout; test_grade_seams.py checks the regex matches it.
# The wrapper line names every test type that holds a ScopedHook member;
# test_grade_seams.py fails when tests/ gains one this line does not name.
for_test name | \b\w+_for_test\b | tests/ui/uitest_details.cpp | : seam(hydra::ui::set_view_analyzer_for_test,
set_fetcher | \bset_fetcher\b | tests/test_update_state.cpp | fetcher(hydra::net::set_fetcher,
set_downloader | \bset_downloader\b | tests/test_update_state.cpp | downloader(hydra::net::set_downloader,
ScopedHook | \bScopedHook\b | tests/ui/uitest_details.cpp | ScopedHook seam;
set_audio_device_factory | \bset_audio_device_factory\b | tests/test_preview_controller.cpp | pc.set_audio_device_factory(no_device_factory());
selected-> assignment | \bselected->\w+(?:(?:\.|->)\w+|\[[^\]]*\])*\s*=(?!=) | tests/ui/uitest_details.cpp | h.app->selected->artist = "Green Day";
seam wrapper object | ^\s*(?:const\s+)?(?:BatchGate|ViewGate|OpenGate|ScanHold|ReportHold|SaveGate|UpdateWorld|UpdateSeams|FailingAnalyzer)\s+\w+\s*(?:[({;]|$) | tests/ui/uitest_details.cpp | ViewGate gate;
